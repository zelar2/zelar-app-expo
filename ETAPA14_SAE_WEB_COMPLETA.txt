type SaeStatus = "aberto" | "em_andamento" | "concluido" | "cancelado";

interface VitalSigns {
  pa?: string;
  fc?: string;
  fr?: string;
  temp?: string;
  spo2?: string;
  dor?: string;
  glicemia?: string;
}

interface Diagnostico {
  titulo: string;
  relacionado_a?: string;
  evidenciado_por?: string;
}

interface Intervencao {
  descricao: string;
  frequencia?: string;
}

interface Prescricao {
  item: string;
  via?: string;
  horario?: string;
  observacao?: string;
  feito?: boolean;
}

interface SaeRecord {
  id: string;
  patient_id: string;
  professional_id: string;
  title: string;
  status: SaeStatus;
  historico: { queixa?: string; anamnese?: string; exame_fisico?: string };
  diagnosticos: Diagnostico[];
  planejamento: Intervencao[];
  prescricoes: Prescricao[];
  avaliacao: string | null;
  vital_signs: VitalSigns;
  opened_at: string;
  closed_at: string | null;
}

interface Evolucao {
  id: string;
  sae_record_id: string;
  patient_id: string;
  professional_id: string;
  content: string;
  vital_signs: VitalSigns;
  procedures: { name: string }[];
  created_at: string;
}

interface PatientOption {
  id: string;
  full_name: string | null;
}

const STATUS_LABEL: Record<SaeStatus, string> = {
  aberto: "Aberto",
  em_andamento: "Em andamento",
  concluido: "Concluído",
  cancelado: "Cancelado",

const STATUS_COLOR: Record<SaeStatus, string> = {
  aberto: "bg-primary/10 text-primary",
  em_andamento: "bg-warning/10 text-warning",
  concluido: "bg-success/10 text-success",
  cancelado: "bg-muted text-muted-foreground",
};

type SectionKey = "coleta" | "diagnosticos" | "planejamento" | "prescricoes" | "avaliacao";
type VersionStatus = "rascunho" | "publicada";

const SECTION_LABEL: Record<SectionKey, string> = {
  coleta: "Coleta",
  diagnosticos: "Diagnósticos",
  planejamento: "Planejamento",

  avaliacao: "Avaliação",
};

interface SectionVersion {
  id: string;

function EnfermagemPage() {
  const { user, primaryRole, loading: authLoading } = useProfile();
  const [records, setRecords] = useState<SaeRecord[]>([]);
  const [loading, setLoading] = useState(false);
  const [selectedId, setSelectedId] = useState<string | null>(null);
  const [newOpen, setNewOpen] = useState(false);

  const isProfessional = primaryRole === "profissional" || primaryRole === "admin";

  const load = useCallback(async () => {
    if (!user) return;
    setLoading(true);
    const { data, error } = await supabase
      .from("sae_records")
      .select("*")
      .order("opened_at", { ascending: false });
    if (error) toast.error("Erro: " + error.message);
    setRecords((data as unknown as SaeRecord[]) ?? []);
    setLoading(false);
  }, [user]);

  useEffect(() => {
    if (!authLoading) void load();
  }, [authLoading, load]);

  const selected = records.find((r) => r.id === selectedId) ?? null;

  if (selected) {
    return (
      <SaeDetail
        record={selected}
        canEdit={isProfessional && selected.professional_id === user?.id}
        onBack={() => setSelectedId(null)}
        onChanged={load}
      />
    );
  }

  return (
    <AppShell title="Enfermagem — SAE" subtitle="Sistematização da Assistência">
      <div className="rounded-3xl bg-gradient-success p-5 text-success-foreground shadow-elevated">
        <p className="text-xs font-medium uppercase tracking-wider text-white/85">SAE</p>
        <h2 className="mt-1 text-xl font-bold">Processo de Enfermagem</h2>
        <p className="mt-1 text-sm text-white/90">
          Coleta · Diagnóstico · Planejamento · Prescrição · Implementação · Evolução
        </p>
        {isProfessional && (
          <Button
            className="mt-4 h-10 rounded-xl bg-white text-success hover:bg-white/90"
            onClick={() => setNewOpen(true)}
          >
            <Plus className="h-4 w-4" /> Iniciar novo processo
          </Button>
        )}
      </div>

      <h3 className="mb-3 mt-6 text-sm font-semibold uppercase tracking-wider text-muted-foreground">
        Meus processos ({records.length})
      </h3>

      {loading ? (
        <div className="flex justify-center py-10">
          <Loader2 className="h-6 w-6 animate-spin text-primary" />
        </div>
      ) : records.length === 0 ? (
        <div className="rounded-2xl border border-dashed border-border bg-card p-8 text-center">
          <ClipboardList className="mx-auto h-8 w-8 text-muted-foreground" />
          <p className="mt-2 text-sm text-muted-foreground">
            {isProfessional
              ? "Nenhum processo aberto. Inicie o primeiro."
              : "Nenhum processo de enfermagem ainda."}
          </p>
        </div>
      ) : (
        <div className="space-y-2.5">
          {records.map((r) => (
            <button
              key={r.id}
              onClick={() => setSelectedId(r.id)}
              className="w-full rounded-2xl border border-border bg-card p-4 text-left shadow-sm hover:shadow-elegant transition-shadow"
            >
              <div className="flex items-start justify-between gap-2">
                <div className="min-w-0 flex-1">
                  <p className="font-semibold text-foreground truncate">{r.title}</p>
                  <p className="mt-1 text-xs text-muted-foreground">
                    Aberto em {new Date(r.opened_at).toLocaleDateString("pt-BR")}
                  </p>
                  <div className="mt-2 flex flex-wrap gap-1.5">
                    <Badge variant="secondary" className="text-[10px]">
                      {r.diagnosticos?.length ?? 0} dx
                    </Badge>
                    <Badge variant="secondary" className="text-[10px]">
                      {r.prescricoes?.length ?? 0} presc
                    </Badge>
                  </div>
                </div>
                <span
                  className={cn(
                    "shrink-0 rounded-full px-2 py-0.5 text-[10px] font-semibold",
                    STATUS_COLOR[r.status],
                  )}
                >
                  {STATUS_LABEL[r.status]}
                </span>
              </div>
            </button>
          ))}
        </div>
      )}

      <NewSaeDialog
        open={newOpen}
        onOpenChange={setNewOpen}
        professionalId={user?.id}
        onCreated={async (id) => {
          setNewOpen(false);
          await load();
          setSelectedId(id);
        }}
      />
    </AppShell>
  );
}

function NewSaeDialog({
  open,
  onOpenChange,
  professionalId,
  onCreated,
}: {
  open: boolean;
  onOpenChange: (v: boolean) => void;
  professionalId?: string;
  onCreated: (id: string) => void | Promise<void>;
}) {
  const [title, setTitle] = useState("");
  const [patientId, setPatientId] = useState<string>("");
  const [search, setSearch] = useState("");
  const [patients, setPatients] = useState<PatientOption[]>([]);
  const [searching, setSearching] = useState(false);
  const [saving, setSaving] = useState(false);

  useEffect(() => {
    if (!open) {
      setTitle("");
      setPatientId("");
      setSearch("");
      setPatients([]);
    }
  }, [open]);

  useEffect(() => {
    if (!open) return;
    const q = search.trim();
    if (q.length < 2) {
      setPatients([]);
      return;
    }
    let cancelled = false;
    setSearching(true);
    const t = setTimeout(async () => {
      const { data } = await supabase
        .from("profiles")
        .select("id, full_name")
        .ilike("full_name", `%${q}%`)
        .limit(10);
      if (cancelled) return;
      setPatients((data as PatientOption[] | null) ?? []);
      setSearching(false);
    }, 250);
    return () => {
      cancelled = true;
      clearTimeout(t);
    };
  }, [search, open]);

  const submit = async () => {
    if (!professionalId || !title.trim() || !patientId) {
      toast.error("Informe título e paciente.");
      return;
    }
    setSaving(true);
    const { data, error } = await supabase
      .from("sae_records")
      .insert({
        title: title.trim(),
        patient_id: patientId,
        professional_id: professionalId,
        status: "aberto" as SaeStatus,
      })
      .select("id")
      .single();
    setSaving(false);
    if (error) return toast.error(error.message);
    toast.success("Processo iniciado!");
    await onCreated(data.id);
  };

  const selectedPatient = patients.find((p) => p.id === patientId);

  return (
    <Dialog open={open} onOpenChange={onOpenChange}>
      <DialogContent className="max-w-md">
        <DialogHeader>
          <DialogTitle>Novo processo de enfermagem</DialogTitle>
        </DialogHeader>
        <div className="space-y-3">
          <div>
            <Label>Título</Label>
            <Input
              value={title}
              onChange={(e) => setTitle(e.target.value)}
              placeholder="Ex: Cuidado domiciliar — Sra. Maria"
            />
          </div>
          <div>
            <Label>Paciente</Label>
            {selectedPatient ? (
              <div className="mt-1 flex items-center gap-2 rounded-xl border border-border bg-card p-2">
                <div className="flex h-8 w-8 items-center justify-center rounded-full bg-primary/10 text-primary">
                  <UserIcon className="h-4 w-4" />
                </div>
                <span className="flex-1 text-sm font-medium">
                  {selectedPatient.full_name ?? "Sem nome"}
                </span>
                <button
                  onClick={() => {
                    setPatientId("");
                    setSearch("");
                  }}
                  className="rounded-lg p-1 text-muted-foreground hover:text-foreground"
                >
                  <X className="h-4 w-4" />
                </button>
              </div>
            ) : (
              <>
                <div className="relative">
                  <Search className="absolute left-3 top-1/2 h-4 w-4 -translate-y-1/2 text-muted-foreground" />
                  <Input
                    className="pl-9"
                    placeholder="Buscar paciente pelo nome"
                    value={search}
                    onChange={(e) => setSearch(e.target.value)}
                  />
                </div>
                {searching && <p className="mt-1 text-xs text-muted-foreground">Buscando...</p>}
                {patients.length > 0 && (
                  <div className="mt-2 max-h-40 space-y-1 overflow-auto rounded-xl border border-border p-1">
                    {patients.map((p) => (
                      <button
                        key={p.id}
                        onClick={() => setPatientId(p.id)}
                        className="flex w-full items-center gap-2 rounded-lg p-2 text-left text-sm hover:bg-muted"
                      >
                        <UserIcon className="h-4 w-4 text-muted-foreground" />
                        {p.full_name ?? "Sem nome"}
                      </button>
                    ))}
                  </div>
                )}
              </>
            )}
          </div>
        </div>
        <DialogFooter>
          <Button variant="ghost" onClick={() => onOpenChange(false)} disabled={saving}>
            Cancelar
          </Button>
          <Button className="bg-gradient-primary" onClick={submit} disabled={saving}>
            {saving && <Loader2 className="h-4 w-4 animate-spin" />}
            Criar
          </Button>
        </DialogFooter>
      </DialogContent>
    </Dialog>
  );
}

function SaeDetail({
  record,
  canEdit,
  onBack,
  onChanged,
}: {
  record: SaeRecord;
  canEdit: boolean;
  onBack: () => void;
  onChanged: () => void | Promise<void>;
}) {
  const [tab, setTab] = useState("coleta");
  const [evolutions, setEvolutions] = useState<Evolucao[]>([]);
  const [loadingEv, setLoadingEv] = useState(false);
  const [evOpen, setEvOpen] = useState(false);

  // Editable local state
  const [historico, setHistorico] = useState(record.historico ?? {});
  const [vital, setVital] = useState<VitalSigns>(record.vital_signs ?? {});
  const [diagnosticos, setDiagnosticos] = useState<Diagnostico[]>(record.diagnosticos ?? []);
  const [planejamento, setPlanejamento] = useState<Intervencao[]>(record.planejamento ?? []);
  const [prescricoes, setPrescricoes] = useState<Prescricao[]>(record.prescricoes ?? []);
  const [avaliacao, setAvaliacao] = useState(record.avaliacao ?? "");
  const [status, setStatus] = useState<SaeStatus>(record.status);
  const [saving, setSaving] = useState(false);

  const loadEvolutions = useCallback(async () => {
    setLoadingEv(true);
    const { data } = await supabase
      .from("sae_evolutions")
      .select("*")
      .eq("sae_record_id", record.id)
      .order("created_at", { ascending: false });
    setEvolutions((data as unknown as Evolucao[]) ?? []);
    setLoadingEv(false);
  }, [record.id]);

  useEffect(() => {
    void loadEvolutions();
  }, [loadEvolutions]);

  const sectionPayload = useCallback(
    (section: SectionKey): Record<string, unknown> => {
      switch (section) {
        case "coleta":
          return { historico, vital_signs: vital };
        case "diagnosticos":
          return { diagnosticos };
        case "planejamento":
          return { planejamento };
        case "prescricoes":
          return { prescricoes };
        case "avaliacao":
          return { avaliacao: avaliacao.trim() || null };
      }
    },
    [historico, vital, diagnosticos, planejamento, prescricoes, avaliacao],
  );

  const applySectionPayload = useCallback(
    (section: SectionKey, payload: Record<string, unknown>) => {
      switch (section) {
        case "coleta":
          setHistorico((payload.historico as SaeRecord["historico"]) ?? {});
          setVital((payload.vital_signs as VitalSigns) ?? {});
          break;
        case "diagnosticos":
          setDiagnosticos((payload.diagnosticos as Diagnostico[]) ?? []);
          break;
        case "planejamento":
          setPlanejamento((payload.planejamento as Intervencao[]) ?? []);
          break;
        case "prescricoes":
          setPrescricoes((payload.prescricoes as Prescricao[]) ?? []);
          break;
        case "avaliacao":
          setAvaliacao(((payload.avaliacao as string | null) ?? "") || "");
          break;
      }
    },
    [],
  );

  const publishSection = useCallback(
    async (section: SectionKey) => {
      const payload = sectionPayload(section);
      const { error } = await supabase
        .from("sae_records")
        .update(payload as never)
        .eq("id", record.id);
      if (error) throw new Error(error.message);
      await onChanged();
    },
    [record.id, sectionPayload, onChanged],
  );

  const updateStatus = async (next: SaeStatus) => {
    setStatus(next);
    setSaving(true);
    const patch: Record<string, unknown> = { status: next };
    if (next === "concluido" && !record.closed_at) patch.closed_at = new Date().toISOString();
    const { error } = await supabase
      .from("sae_records")
      .update(patch as never)
      .eq("id", record.id);
    setSaving(false);
    if (error) return toast.error(error.message);
    await onChanged();
  };

  const del = async () => {
    if (!confirm("Excluir este processo?")) return;
    const { error } = await supabase.from("sae_records").delete().eq("id", record.id);
    if (error) return toast.error(error.message);
    toast.success("Processo excluído");
    onBack();

    onBack();
    await onChanged();
  };

  return (
    <AppShell
      title={record.title}
      subtitle={`Aberto em ${new Date(record.opened_at).toLocaleDateString("pt-BR")}`}
      headerLeft={
        <button onClick={onBack} className="rounded-lg p-2 text-foreground">
          <ArrowLeft className="h-5 w-5" />
        </button>
      }
    >
      <div className="mb-4 flex items-center gap-2">
        <span
          className={cn("rounded-full px-2.5 py-1 text-xs font-semibold", STATUS_COLOR[status])}

            value={status}

            disabled={saving}

          />

          {planejamento.map((p, i) => (
            <div key={i} className="rounded-2xl border border-border bg-card p-3 shadow-sm">

        {/* 4. Prescrições */}

          ))}

            <p className="text-center text-sm text-muted-foreground">Sem prescrições.</p>
          )}
        </TabsContent>

        {/* 5. Evoluções */}
        <TabsContent value="evolucao" className="mt-4 space-y-3">
          {canEdit && (

              Nenhuma evolução registrada.
            </p>
          ) : (
            evolutions.map((e) => (
              <div key={e.id} className="rounded-2xl border border-border bg-card p-3 shadow-sm">
                <p className="text-[11px] text-muted-foreground">
                  {new Date(e.created_at).toLocaleString("pt-BR")}
                </p>
                <p className="mt-1 whitespace-pre-wrap text-sm text-foreground">{e.content}</p>
                {Object.keys(e.vital_signs ?? {}).length > 0 && (
                  <div className="mt-2 flex flex-wrap gap-1.5">
                    {Object.entries(e.vital_signs).map(([k, v]) =>
                      v ? (
                        <Badge key={k} variant="secondary" className="text-[10px]">
                          {k.toUpperCase()}: {v as string}
                        </Badge>
                      ) : null,
                    )}
                  </div>
                )}
              </div>
            ))
          )}
          <NewEvolucaoDialog
            open={evOpen}
            onOpenChange={setEvOpen}
            record={record}
            onSaved={async () => {
              setEvOpen(false);
              await loadEvolutions();
              toast.success("Evolução registrada");
            }}
          />
        </TabsContent>

        {/* 6. Avaliação final */}
        <TabsContent value="avaliacao" className="mt-4 space-y-3">
          <SectionVersionBar
            recordId={record.id}
            section="avaliacao"
            canEdit={canEdit}
            currentPayload={sectionPayload("avaliacao")}
            onRestore={(p) => applySectionPayload("avaliacao", p)}
            onPublish={() => publishSection("avaliacao")}
          />

          <Field label="Avaliação e desfecho">
            <Textarea
              rows={8}
              value={avaliacao}
              readOnly={!canEdit}
              placeholder="Descreva a resposta do paciente às intervenções, evolução do quadro e desfecho."
              onChange={(e) => setAvaliacao(e.target.value)}
            />
          </Field>
        </TabsContent>
      </Tabs>
    </AppShell>
  );
}

function NewEvolucaoDialog({
  open,
  onOpenChange,
  record,
  onSaved,
}: {
  open: boolean;
  onOpenChange: (v: boolean) => void;
  record: SaeRecord;
  onSaved: () => void | Promise<void>;
}) {
  const [content, setContent] = useState("");
  const [vital, setVital] = useState<VitalSigns>({});
  const [saving, setSaving] = useState(false);

  useEffect(() => {
    if (!open) {
      setContent("");
      setVital({});
    }
  }, [open]);

  const submit = async () => {
    if (!content.trim()) {
      toast.error("Descreva a evolução.");
      return;
    }
    setSaving(true);

      patient_id: record.patient_id,
      professional_id: record.professional_id,
      content: content.trim(),
      vital_signs: vital as never,
      procedures: [],
    });

        patient_id: record.patient_id,
        author_id: record.professional_id,
        category: "evolucao",
        title: `Evolução SAE — ${record.title}`,
        content: content.trim(),
        data: { vital_signs: vital, sae_record_id: record.id } as never,
      });
    }
    setSaving(false);
    if (error) return toast.error(error.message);
    await onSaved();

  return (
    <Dialog open={open} onOpenChange={onOpenChange}>
      <DialogContent className="max-w-md">
        <DialogHeader>
          <DialogTitle>Nova evolução</DialogTitle>
        </DialogHeader>

          <div>

              rows={5}

function SectionVersionBar({
  recordId,
  section,
  canEdit,
  currentPayload,
  onRestore,
  onPublish,
}: {
  recordId: string;
  section: SectionKey;
  canEdit: boolean;
  currentPayload: Record<string, unknown>;
  onRestore: (payload: Record<string, unknown>) => void;
  onPublish: () => Promise<void>;
}) {
  const { user } = useProfile();
  const [versions, setVersions] = useState<SectionVersion[]>([]);
  const [loading, setLoading] = useState(false);
  const [saving, setSaving] = useState<null | "rascunho" | "publicada">(null);
  const [histOpen, setHistOpen] = useState(false);
  const [note, setNote] = useState("");

  const load = useCallback(async () => {
    setLoading(true);
    const { data, error } = await supabase
      .from("sae_section_versions")
      .select("*")
      .eq("sae_record_id", recordId)
      .eq("section", section)
      .order("created_at", { ascending: false });
    if (error) toast.error(error.message);
    setVersions((data as unknown as SectionVersion[]) ?? []);
    setLoading(false);
  }, [recordId, section]);

  useEffect(() => {
    load();
  }, [load]);

  const latestDraft = versions.find((v) => v.status === "rascunho") ?? null;
  const latestPublished = versions.find((v) => v.status === "publicada") ?? null;
  const publishedCount = versions.filter((v) => v.status === "publicada").length;

  const nextVersion = (status: VersionStatus) => {
    if (status === "publicada") return publishedCount + 1;
    return (latestDraft?.version_number ?? 0) + 1;
  };

  const create = async (status: VersionStatus) => {
    if (!user) return;
    setSaving(status);
    try {
      if (status === "publicada") {
        await onPublish();
      }
      const { error } = await supabase.from("sae_section_versions").insert({
        sae_record_id: recordId,
        section,
        status,
        version_number: nextVersion(status),
        payload: currentPayload,
        note: note.trim() || null,
        author_id: user.id,
      } as never);
      if (error) throw new Error(error.message);
      setNote("");
      toast.success(status === "rascunho" ? "Rascunho salvo" : "Versão publicada");
      await load();
    } catch (e) {
      toast.error(e instanceof Error ? e.message : "Erro ao salvar");
    } finally {
      setSaving(null);
    }
  };

  const remove = async (id: string) => {
    if (!confirm("Excluir esta versão?")) return;
    const { error } = await supabase.from("sae_section_versions").delete().eq("id", id);
    if (error) return toast.error(error.message);
    toast.success("Versão excluída");
    await load();
  };

  return (
    <>
      <div className="rounded-2xl border border-border bg-card p-3 shadow-sm">
        <div className="flex flex-wrap items-center gap-2">
          <FileClock className="h-4 w-4 text-primary" />
          <span className="text-xs font-semibold text-foreground">{SECTION_LABEL[section]}</span>
          {latestPublished ? (
            <Badge variant="secondary" className="text-[10px]">
              v{latestPublished.version_number} publicada ·{" "}
              {new Date(latestPublished.created_at).toLocaleDateString("pt-BR")}
            </Badge>
          ) : (
